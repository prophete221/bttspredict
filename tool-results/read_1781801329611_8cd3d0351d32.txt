   326→// Generate a deterministic daily cote between 15 and 30 based on the date
   327→function getDailyCote(): number {
   328→  const today = new Date()
   329→  const seed = today.getFullYear() * 10000 + (today.getMonth() + 1) * 100 + today.getDate()
   330→  // Simple deterministic pseudo-random from seed
   331→  const x = Math.sin(seed * 9301 + 49297) * 233280
   332→  const fraction = x - Math.floor(x)
   333→  return Math.round((15 + fraction * 15) * 100) / 100 // between 15.00 and 30.00
   334→}
