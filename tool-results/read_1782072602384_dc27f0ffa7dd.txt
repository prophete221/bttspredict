    55→function formatDateShort(dateStr: string) {
    56→  if (!dateStr) return ''
    57→  try {
    58→    const d = new Date(dateStr + 'T12:00:00')
    59→    const day = d.getDate().toString().padStart(2, '0')
    60→    const month = (d.getMonth() + 1).toString().padStart(2, '0')
    61→    const today = new Date()
    62→    today.setHours(12, 0, 0, 0)
    63→    const matchDate = new Date(dateStr + 'T12:00:00')
    64→    const diffDays = Math.round((matchDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24))
    65→
    66→    if (diffDays === 0) return `Auj.`
    67→    if (diffDays === 1) return `Dem.`
    68→    const weekdays = ['Dim', 'Lun', 'Mar', 'Mer', 'Jeu', 'Ven', 'Sam']
    69→    const weekday = weekdays[d.getDay()]
    70→    return `${weekday} ${day}/${month}`
    71→  } catch {
    72→    return dateStr
    73→  }
    74→}

    76→function getDateLabel(dateStr: string) {
    77→  if (!dateStr) return 'upcoming'
    78→  const today = new Date()
    79→  today.setHours(12, 0, 0, 0)
    80→  const matchDate = new Date(dateStr + 'T12:00:00')
    81→  const diffDays = Math.round((matchDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24))
    82→  if (diffDays === 0) return 'today'
    83→  if (diffDays === 1) return 'tomorrow'
    84→  return 'upcoming'
    85→}

   387→export default function FreePredictions() {

   393→
   394→  useEffect(() => {
   395→    fetch('/predictions.json')
   396→      .then((r) => {
   397→        if (!r.ok) throw new Error(`HTTP ${r.status}`)
   398→        return r.json()
   399→      })
   400→      .then((data) => {
   401→        setPredictions(data.predictions || [])
   402→        setLoading(false)
   403→      })
   404→      .catch(() => {
   405→        setPredictions([])
   406→        setError('Impossible de charger les pronostics. Réessayez plus tard.')
   407→        setLoading(false)
   408→      })
   409→  }, [])
   410→
   411→  const groupedPredictions = useMemo(() => {
   412→    return predictions.reduce<Record<string, MatchData>>((groups, p) => {
   413→      const key = `${p.match}|${p.date || ''}|${p.time || ''}`
   414→      if (!groups[key]) {
   415→        groups[key] = {
   416→          match: p.match,
   417→          league: p.league,
   418→          date: p.date,
   419→          time: p.time,
   420→          matchSemantic: p.matchSemantic,
   421→          homeLogo: p.homeLogo || '',
   422→          awayLogo: p.awayLogo || '',
   423→          btts: null,
   424→          over25: null,
   425→        }
   426→      }
   427→      if (p.type === 'BTTS') {
   428→        groups[key].btts = { prediction: p.prediction, confidence: p.confidence }
   429→      } else if (p.type === 'Over 2.5') {
   430→        groups[key].over25 = { prediction: p.prediction, confidence: p.confidence }
   431→      }
   432→      return groups
   433→    }, {})
   434→  }, [predictions])
   435→
   436→  const matchList = useMemo(() => Object.values(groupedPredictions), [groupedPredictions])
   437→
   438→  const leagues = useMemo(() => {
   439→    const leagueSet = new Set(matchList.map(m => m.league).filter(Boolean))
   440→    return ['all', ...Array.from(leagueSet).slice(0, 6)]
   441→  }, [matchList])
   442→
   443→  const filteredMatches = useMemo(() => {
   444→    if (activeLeague === 'all') return matchList
   445→    return matchList.filter(m => m.league === activeLeague)
   446→  }, [matchList, activeLeague])
   447→
   448→  const dateGroups = useMemo(() => {
   449→    const groups: Record<string, MatchData[]> = { today: [], tomorrow: [], upcoming: [] }
   450→    filteredMatches.forEach(m => {
   451→      const label = getDateLabel(m.date)
   452→      groups[label].push(m)
   453→    })
   454→    return groups
   455→  }, [filteredMatches])
   456→
   457→  const stats = useMemo(() => {
   458→    const bttsOui = matchList.filter(m => m.btts?.prediction === 'Oui').length
   459→    const o25Oui = matchList.filter(m => m.over25?.prediction === 'Oui').length
   460→    return { total: matchList.length, bttsOui, o25Oui }
   461→  }, [matchList])
   462→
   463→  // Count-up animations for stats — only mount after predictions are loaded so
   464→  // the hook receives the correct target value.
   465→  const totalStats = !loading ? stats.total : 0
   466→  const bttsStats = !loading ? stats.bttsOui : 0
   467→  const o25Stats = !loading ? stats.o25Oui : 0
