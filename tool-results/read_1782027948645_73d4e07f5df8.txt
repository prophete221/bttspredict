   203→/* ─────────────────────── FIFA POOL (generates teams) ─────────────────────── */
   204→
   205→const FIFA_TEAMS = [
   206→  'France', 'Brazil', 'Argentina', 'Germany', 'Spain', 'England',
   207→  'Portugal', 'Italy', 'Netherlands', 'Belgium', 'Croatia', 'Morocco',
   208→  'Japan', 'South Korea', 'Uruguay', 'Colombia', 'Mexico', 'USA',
   209→  'Senegal', 'Nigeria', 'Cameroon', 'Egypt', 'Ghana', 'Ivory Coast',
   210→  'Denmark', 'Switzerland', 'Austria', 'Serbia', 'Poland', 'Sweden',
   211→  'Norway', 'Scotland', 'Turkey', 'Algeria', 'Tunisia', 'DR Congo',
   212→]
   213→
   214→const FIFA_LEAGUES = [
   215→  'FIFA World Cup', 'Copa America', 'EURO Qualifiers', 'African Cup',
   216→  'Asian Cup', 'Nations League', 'Friendly International', 'World Cup Qualifiers',
   217→]
   218→
   219→function seededRandom(seed: number): number {
   220→  const x = Math.sin(seed * 9301 + 49297) * 233280
   221→  return x - Math.floor(x)
   222→}
   223→
   224→function generateFifaMatches(seed: number, count: number) {
   225→  const matches: Array<{ home: string; away: string; league: string }> = []
   226→  for (let i = 0; i < count; i++) {
   227→    const r1 = seededRandom(seed + i * 7)
   228→    const r2 = seededRandom(seed + i * 13 + 3)
   229→    const r3 = seededRandom(seed + i * 19 + 7)
   230→
   231→    const homeIdx = Math.floor(r1 * FIFA_TEAMS.length)
   232→    let awayIdx = Math.floor(r2 * FIFA_TEAMS.length)
   233→    if (awayIdx === homeIdx) awayIdx = (awayIdx + 1) % FIFA_TEAMS.length
   234→
   235→    const home = FIFA_TEAMS[homeIdx]
   236→    const away = FIFA_TEAMS[awayIdx]
   237→    const league = FIFA_LEAGUES[Math.floor(r3 * FIFA_LEAGUES.length)]
   238→
   239→    matches.push({ home, away, league })
   240→  }
   241→  return matches
   242→}
