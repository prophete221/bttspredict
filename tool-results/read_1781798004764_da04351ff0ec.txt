   285→   285→// Generate a deterministic daily cote between 15 and 30 based on the date
   286→   286→function getDailyCote(): number {
   287→   287→  const today = new Date()
   288→   288→  const seed = today.getFullYear() * 10000 + (today.getMonth() + 1) * 100 + today.getDate()
   289→   289→  // Simple deterministic pseudo-random from seed
   290→   290→  const x = Math.sin(seed * 9301 + 49297) * 233280
   291→   291→  const fraction = x - Math.floor(x)
   292→   292→  return Math.round((15 + fraction * 15) * 100) / 100 // between 15.00 and 30.00
   293→   293→}
