   339→// Generate a deterministic daily cote between 15 and 30 based on the date
   340→function getDailyCote(): number {
   341→  const today = new Date()
   342→  const seed = today.getFullYear() * 10000 + (today.getMonth() + 1) * 100 + today.getDate()
   343→  // Simple deterministic pseudo-random from seed
   344→  const x = Math.sin(seed * 9301 + 49297) * 233280
   345→  const fraction = x - Math.floor(x)
   346→  return Math.round((15 + fraction * 15) * 100) / 100 // between 15.00 and 30.00
