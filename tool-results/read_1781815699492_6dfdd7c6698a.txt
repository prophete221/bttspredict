   338→  const seed = today.getFullYear() * 10000 + (today.getMonth() + 1) * 100 + today.getDate()
   339→  // Simple deterministic pseudo-random from seed
   340→  const x = Math.sin(seed * 9301 + 49297) * 233280
   341→  const fraction = x - Math.floor(x)
   342→  return Math.round((15 + fraction * 15) * 100) / 100 // between 15.00 and 30.00
   343→}
