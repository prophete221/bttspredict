   201→/* ─────────────────────── FIFA POOL (generates teams) ─────────────────────── */
   202→
   203→const FIFA_TEAMS = [
   204→  'France', 'Brazil', 'Argentina', 'Germany', 'Spain', 'England',
   205→  'Portugal', 'Italy', 'Netherlands', 'Belgium', 'Croatia', 'Morocco',
   206→  'Japan', 'South Korea', 'Uruguay', 'Colombia', 'Mexico', 'USA',
   207→  'Senegal', 'Nigeria', 'Cameroon', 'Egypt', 'Ghana', 'Ivory Coast',
   208→  'Denmark', 'Switzerland', 'Austria', 'Serbia', 'Poland', 'Sweden',
   209→  'Norway', 'Scotland', 'Turkey', 'Algeria', 'Tunisia', 'DR Congo',
   210→]
   211→
   212→const FIFA_LEAGUES = [
   213→  'FIFA World Cup', 'Copa America', 'EURO Qualifiers', 'African Cup',
   214→  'Asian Cup', 'Nations League', 'Friendly International', 'World Cup Qualifiers',
   215→]
   216→
   217→function seededRandom(seed: number): number {
   218→  const x = Math.sin(seed * 9301 + 49297) * 233280
   219→  return x - Math.floor(x)
   220→}
   221→
   222→function generateFifaMatches(seed: number, count: number) {
   223→  const matches = []
   224→  for (let i = 0; i < count; i++) {
   225→    const r1 = seededRandom(seed + i * 7)
   226→    const r2 = seededRandom(seed + i * 13 + 3)
   227→    const r3 = seededRandom(seed + i * 19 + 7)
   228→
   229→    const homeIdx = Math.floor(r1 * FIFA_TEAMS.length)
   230→    let awayIdx = Math.floor(r2 * FIFA_TEAMS.length)
   231→    if (awayIdx === homeIdx) awayIdx = (awayIdx + 1) % FIFA_TEAMS.length
   232→
   233→    const home = FIFA_TEAMS[homeIdx]
   234→    const away = FIFA_TEAMS[awayIdx]
   235→    const league = FIFA_LEAGUES[Math.floor(r3 * FIFA_LEAGUES.length)]
   236→
   237→    matches.push({ home, away, league })
   238→  }
   239→  return matches
   240→}
