   291→// Sport-specific team/player logo (initials fallback since real logos are not provided)
   292→function SportTeamLogo({ name, accent = 'gold', size = 18 }: { name: string; accent?: 'gold' | 'emerald' | 'red' | 'cyan'; size?: number }) {
   293→  const initials = name?.split(' ').map(w => w[0]).slice(0, 2).join('').toUpperCase() || '?'
   294→  const accentMap = {
   295→    gold: 'bg-gold/10 border-gold/15 text-gold/70',
   296→    emerald: 'bg-emerald/10 border-emerald/15 text-emerald/70',
   297→    red: 'bg-red-500/10 border-red-500/15 text-red-400',
   298→    cyan: 'bg-cyan-400/10 border-cyan-400/15 text-cyan-400',
   299→  }
   300→  return (
   301→    <div className={`rounded-full border flex items-center justify-center font-bold flex-shrink-0 ${accentMap[accent]}`} style={{ width: size, height: size, fontSize: size * 0.4 }}>
   302→      {initials}
   303→    </div>
   304→  )
   305→}

   474→// Deterministic daily cote per sport
   475→function getSportDailyCote(sportId: string, min: number, max: number): number {
   476→  const today = new Date()
   477→  const seed = today.getFullYear() * 10000 + (today.getMonth() + 1) * 100 + today.getDate() + sportId.length
   478→  const x = Math.sin(seed * 9301 + 49297) * 233280
   479→  const fraction = x - Math.floor(x)
   480→  return Math.round((min + fraction * (max - min)) * 100) / 100
   481→}
