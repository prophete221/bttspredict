    55→function formatDateShort(dateStr: string) {
    56→  if (!dateStr) return ''
    57→  try {
    58→    const d = new Date(dateStr + 'T12:00:00')
    59→    const day = d.getDate().toString().padStart(2, '0')
    60→    const month = (d.getMonth() + 1).toString().padStart(2, '0')
    61→    const today = new Date()
    62→    today.setHours(12, 0, 0, 0)
    63→    const matchDate = new Date(dateStr + 'T12:00:00')
    64→    const diffDays = Math.round((matchDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24))
    65→
    66→    if (diffDays === 0) return `Auj.`
    67→    if (diffDays === 1) return `Dem.`
    68→    const weekdays = ['Dim', 'Lun', 'Mar', 'Mer', 'Jeu', 'Ven', 'Sam']
    69→    const weekday = weekdays[d.getDay()]
    70→    return `${weekday} ${day}/${month}`
    71→  } catch {
    72→    return dateStr
    73→  }
    74→}

    76→function getDateLabel(dateStr: string) {
    77→  if (!dateStr) return 'upcoming'
    78→  const today = new Date()
    79→  today.setHours(12, 0, 0, 0)
    80→  const matchDate = new Date(dateStr + 'T12:00:00')
    81→  const diffDays = Math.round((matchDate.getTime() - today.getTime()) / (1000 * 60 * 60 * 24))
    82→  if (diffDays === 0) return 'today'
    83→  if (diffDays === 1) return 'tomorrow'
    84→  return 'upcoming'
    85→}

   395→
   396→  useEffect(() => {
   397→    fetch('/predictions.json')
   398→      .then((r) => {
   399→        if (!r.ok) throw new Error(`HTTP ${r.status}`)
   400→        return r.json()
   401→      })
   402→      .then((data) => {
   403→        setPredictions(data.predictions || [])
   404→        setLoading(false)
   405→      })
   406→      .catch(() => {
   407→        setPredictions([])
   408→        setError('Impossible de charger les pronostics. Réessayez plus tard.')
   409→        setLoading(false)
   410→      })
   411→  }, [])
   412→
   413→  const groupedPredictions = useMemo(() => {
   414→    return predictions.reduce<Record<string, MatchData>>((groups, p) => {
   415→      const key = `${p.match}|${p.date || ''}|${p.time || ''}`
   416→      if (!groups[key]) {
   417→        groups[key] = {
   418→          match: p.match,
   419→          league: p.league,
   420→          date: p.date,
   421→          time: p.time,
   422→          matchSemantic: p.matchSemantic,
   423→          homeLogo: p.homeLogo || '',
   424→          awayLogo: p.awayLogo || '',
   425→          btts: null,
   426→          over25: null,
   427→        }
   428→      }
   429→      if (p.type === 'BTTS') {
   430→        groups[key].btts = { prediction: p.prediction, confidence: p.confidence }
   431→      } else if (p.type === 'Over 2.5') {
   432→        groups[key].over25 = { prediction: p.prediction, confidence: p.confidence }
   433→      }
   434→      return groups
   435→    }, {})
   436→  }, [predictions])
   437→
   438→  const matchList = useMemo(() => Object.values(groupedPredictions), [groupedPredictions])
   439→
   440→  const leagues = useMemo(() => {
   441→    const leagueSet = new Set(matchList.map(m => m.league).filter(Boolean))
   442→    return ['all', ...Array.from(leagueSet).slice(0, 6)]
   443→  }, [matchList])
   444→
   445→  const filteredMatches = useMemo(() => {
   446→    if (activeLeague === 'all') return matchList
   447→    return matchList.filter(m => m.league === activeLeague)
   448→  }, [matchList, activeLeague])
   449→
   450→  const dateGroups = useMemo(() => {
   451→    const groups: Record<string, MatchData[]> = { today: [], tomorrow: [], upcoming: [] }
   452→    filteredMatches.forEach(m => {
   453→      const label = getDateLabel(m.date)
   454→      groups[label].push(m)
   455→    })
   456→    return groups
   457→  }, [filteredMatches])
   458→
   459→  const stats = useMemo(() => {
   460→    const bttsOui = matchList.filter(m => m.btts?.prediction === 'Oui').length
   461→    const o25Oui = matchList.filter(m => m.over25?.prediction === 'Oui').length
   462→    return { total: matchList.length, bttsOui, o25Oui }
   463→  }, [matchList])
   464→
   465→  // Count-up animations for stats — only mount after predictions are loaded so
   466→  // the hook receives the correct target value.
   467→  const totalStats = !loading ? stats.total : 0
