   219→   219→function seededRandom(seed: number): number {
   220→   220→  const x = Math.sin(seed * 9301 + 49297) * 233280
   221→   221→  return x - Math.floor(x)
   222→   222→}
   223→   223→
   224→   224→function generateFifaMatches(seed: number, count: number) {
   225→   225→  const matches: Array<{ home: string; away: string; league: string }> = []
   226→   226→  for (let i = 0; i < count; i++) {
   227→   227→    const r1 = seededRandom(seed + i * 7)
   228→   228→    const r2 = seededRandom(seed + i * 13 + 3)
   229→   229→    const r3 = seededRandom(seed + i * 19 + 7)
   230→   230→
   231→   231→    const homeIdx = Math.floor(r1 * FIFA_TEAMS.length)
   232→   232→    let awayIdx = Math.floor(r2 * FIFA_TEAMS.length)
   233→   233→    if (awayIdx === homeIdx) awayIdx = (awayIdx + 1) % FIFA_TEAMS.length
   234→   234→
   235→   235→    const home = FIFA_TEAMS[homeIdx]
   236→   236→    const away = FIFA_TEAMS[awayIdx]
   237→   237→    const league = FIFA_LEAGUES[Math.floor(r3 * FIFA_LEAGUES.length)]
   238→   238→
   239→   239→    matches.push({ home, away, league })
   240→   240→  }
   241→   241→  return matches
   242→   242→}
